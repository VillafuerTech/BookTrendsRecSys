\begin{frame}{The Problem}
\begin{itemize}
  \item \textbf{Too many options}: for most readers, choosing a next book takes time.
  \item \textbf{Not sure what to trust}: ratings are messy and popularity can be misleading.
  \item \textbf{Staff time is limited}: librarians and curators need quick shortlists.
\end{itemize}
\end{frame}

% ================================================================
\begin{frame}{Our Promise}
\begin{itemize}
  \item \textbf{A clean Top-10 list} per user --- titles you can actually act on.
  \item \textbf{At a glance insights}: why these books appear (based on past likes).
  \item \textbf{Fair experience}: we watch for gaps across regions and genres.
\end{itemize}
\end{frame}

% ================================================================
\begin{frame}{What You Get (Benefits)}
\begin{itemize}
  \item \textbf{Save time}: fewer clicks, faster choices.
  \item \textbf{Better fit}: suggestions follow what the reader enjoyed before.
  \item \textbf{Explainable}: simple visuals to see patterns in ratings.
  \item \textbf{Portable}: runs as a small app; no special hardware needed.
\end{itemize}
\end{frame}

% ================================================================
\begin{frame}{One-Minute View of How It Works}
\begin{block}{Behind the scenes}
  \small Your past ratings $\rightarrow$ we learn look-alike tastes $\rightarrow$ score many books $\rightarrow$ show the Top-10.\par
\end{block}
\begin{itemize}
  \item Think of it as ``people who liked X also liked Y'' made practical.
  \item If the list is empty, we fall back to helpful, popular picks.
\end{itemize}
\end{frame}

\begin{itemize}
  \item Works best if the reader has \textbf{some} rating history.
  \item Sparse genres may need \textbf{fallback} to popularity.
  \item We focus on relevance first; \textbf{deep explanations} are in progress.
\end{itemize}

\begin{frame}{(Backup) Ranking Metrics --- Formulas}
\footnotesize
For user \(u\) with ground truth \(G_u\) and top-\(K\) list \(R_u=[i_1,\dots,i_K]\):
\[
\mathrm{P@K}=\frac{1}{|U|}\sum_{u}\frac{|R_u\cap G_u|}{K},\quad
\mathrm{R@K}=\frac{1}{|U|}\sum_{u}\frac{|R_u\cap G_u|}{|G_u|}
\]
\[
\mathrm{DCG@K}(u)=\sum_{j=1}^{K}\frac{\mathbf{1}[i_j\in G_u]}{\log_2(j+1)},\quad
\mathrm{NDCG@K}=\frac{1}{|U|}\sum_{u}\frac{\mathrm{DCG@K}(u)}{\mathrm{IDCG@K}(u)}
\]
\end{frame}
